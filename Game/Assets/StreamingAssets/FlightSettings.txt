# --- 加速 ---------------------------------------------------
# 上下左右スラストの加速度。大きいほど横移動の立ち上がりが速い。
LateralThrust = 40

# RT/LT(前進/後退)の最大加速度。トリガーを最後まで引いたときの値で、
# 押し込み量(0〜1)に比例して 0〜ForwardThrust の範囲で加速度が変わる。
ForwardThrust = 60

# トリガーの遊び(0〜0.9)。この値以下の押し込みは無視される。誤入力防止用。
TriggerDeadZone = 0.05

# 最高速度。慣性で溜められる速度の上限。
MaxSpeed = 60

# --- 慣性 ---------------------------------------------------
# 入力を離したときの減速度。
#   小さいほど長く滑る(慣性が強い)。例: 3 = かなり滑る / 8 = 標準 / 30 = ほぼ即停止
#   0 にすると永遠に止まらない。
CoastDeceleration = 8

# --- 視点 ---------------------------------------------------
# 右スティックを最大に倒したときの旋回速度(度/秒)。
# 倒し量に応じて 0〜この値の範囲でアナログに変わる。
MaxTurnRate = 150

# 右スティックの遊び(0〜0.9)。この値以下の倒し量は無視される。
LookDeadZone = 0.1

# 倒し量に対する応答カーブの指数。
#   1 = 線形 / 大きいほど小さな倒しが繊細(照準向き)で、最大付近で急に回る。
LookResponseCurve = 1.5

# 機体のバンク(傾き)の追従速度。大きいほど素早く傾く。
RollLerp = 6

# 横移動時の最大バンク角(度)。
BankOnStrafeDeg = 25
